hostname EDGE
no ip domain lookup
ipv6 unicast-routing
line con 0
 exec-timeout 0 0
 logging synchronous
interface Loopback0
 ip address 10.255.255.2 255.255.255.255
 ipv6 address 2001:db8:acad:ffff::2/128
 no shutdown
interface Ethernet0/0
 description Link para CORE1
 ip address 10.0.13.2 255.255.255.252
 ipv6 address 2001:db8:acad:013::2/64
 no shutdown
interface Ethernet0/1
 description Link para CORE2
 ip address 10.0.23.2 255.255.255.252
 ipv6 address 2001:db8:acad:023::2/64
 no shutdown
interface Ethernet0/2
 description Link para ISP
 ip address 203.0.113.1 255.255.255.252
 ipv6 address 2001:db8:ffff::1/64
 no shutdown
ip route 0.0.0.0 0.0.0.0 203.0.113.2
ipv6 route ::/0 2001:db8:ffff::2
router eigrp EMPRESA
 address-family ipv4 unicast autonomous-system 100
  topology base
   redistribute static
  exit-af-topology
  network 10.0.0.0
  eigrp router-id 3.3.3.3
 exit-address-family
 address-family ipv6 unicast autonomous-system 100
  af-interface Ethernet0/2
   shutdown
  exit-af-interface
  topology base
   redistribute static
  exit-af-topology
  eigrp router-id 3.3.3.3
 exit-address-family
end
